function doPost(e) { var d = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("الجرد") || ss.insertSheet("الجرد"); if (sh.getLastRow() === 0) { sh.appendRow(["الاسم","العدد","التعبئة","الكارتون","الباركود","الصورة","تم الجرد","التاريخ"]); } if (d.action === "add") { sh.appendRow([d.name, d.qty, d.fill, d.carton, d.barcode||"", d.image||"", false, new Date().toLocaleString('ar-SA')]); return ok(); } if (d.action === "delete") { var rows = sh.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][0] === d.name) { sh.deleteRow(i + 1); break; } } return ok(); } if (d.action === "update") { var rows = sh.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] === d.name) { sh.getRange(i + 1, 7).setValue(d.done); break; } } return ok(); } function ok() { return ContentService.createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); } } function doGet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName("الجرد"); if (!sh) return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON); var rows = sh.getDataRange().getValues(); var r = []; for (var i = 1; i < rows.length; i++) { r.push({ name: rows[i][0], qty: rows[i][1], fill: rows[i][2], carton: rows[i][3], barcode: rows[i][4], image: rows[i][5], done: rows[i][6], date: rows[i][7] }); } return ContentService.createTextOutput(JSON.stringify(r)) .setMimeType(ContentService.MimeType.JSON); }
